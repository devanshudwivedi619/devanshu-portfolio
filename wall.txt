def sales_summary(rows):
    total_sales = sum(row['sales'] for row in rows)
    total_profit = sum(row['profit'] for row in rows)
    total_orders = len({row['order_id'] for row in rows})
    return {
        'sales': total_sales,
        'profit': total_profit,
        'orders': total_orders,
    }
def prepare_bookings(df):
    df = df.drop_duplicates().copy()
    df['children'] = df['children'].fillna(0)
    df['country'] = df['country'].fillna('Unknown')
    df['total_nights'] = (
        df['stays_in_weekend_nights']
        + df['stays_in_week_nights']
    )
    return df
const utilizationRate = (used, purchased) => {
  if (!purchased) return 0;
  return (used / purchased) * 100;
};

const unusedLicenses = (purchased, used) =>
  Math.max(0, purchased - used);
